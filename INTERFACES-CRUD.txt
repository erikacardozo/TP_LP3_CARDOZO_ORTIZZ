public interface Usuario{
	//POST - crear usuario
	createUsuario();
	
	//DELETE - elimina usuario
	deleteUsuario();

	//PUT - actualizar usuario
	modifyUsuario();

	//GET - listar todos los usuarios
	List<Usuarios> (String nombre, String apellido, Long CI, Integer ID);

}

public interface TiposImpuestos {
	//POST - ingresar IVA
	addIVA();
	//POST - ingresar IRP
	addIRP();

}

public interface Mercosur{
	//GET - trae los impuestos del IVA
	getIVA();

	//GET - trae los impuestos del IRP
	getIRP();

	//calcula el descuento al ser del mercosur
	getDescuento(); 
}

public interface Extrazona{
	//GET - trae los impuestos del IVA
	getIVA();
	
//GET - trae los impuestos del IRP
	getIRP();

	//calcula el descuento al ser del Extrazona
	getDescuento(); 
}
